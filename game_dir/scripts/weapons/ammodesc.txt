ammoinfo					// begin ammo definition
{
	// RTN: era "barret" (nome herdado do Paranoia2 original - confirmado em
	// dlls/hgrunt.cpp de la, nao e erro deste port) para o 9mm comum da Glock.
	// Renomeado pro calibre real (9x19mm Parabellum) e dano recalibrado por
	// energia de boca real (~500J, o mais fraco da tabela) - estava em 17/12,
	// quase nivel de fuzil, o mais alto entre os calibres de pistola.
	name		"9mm"		// unique name of ammotype
	MaxCarry		60		// max carry for players
	PlayerDamage	12		// damage if bullet was fired by player (affected to monsters)
	MonsterDamage	7		// damage if bullet was fired by monster (affected to client in singleplayer)
	ShellModel	"models/glock_shell.mdl"
}

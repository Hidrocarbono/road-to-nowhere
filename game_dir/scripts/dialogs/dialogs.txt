//  Parser: server/dialogscript.cpp (server-only, mesmo motivo de
//  weaponscript.cpp - o cliente nao pode saber o proximo no de antemao).
//
//  Todo TEXTO (npc_line, option, exit_option, sealed_line) e uma CHAVE de
//  game_dir/titles.txt, nunca o texto em si.
// ============================================================================

// NOTA: usamos "medico", nao "doutor", porque game_dir/titles.txt JA TINHA
// uma entrada de exemplo chamada "Doutor_Intro" (o "Dr. Freeman" de
// demonstracao, sem relacao com este NPC) - reusar o nome ia colidir com
// texto de outro contexto. Ver game_dir/devkit/GUIA_DIALOGOS.md.

dialog_medico
{
	repeatable    0
	sealed_line   "Medico_JaConversamos"

	npc_line      "Medico_Intro"
	option 1      "Opcao_PerguntarVacina" -> dialog_medico_vacina
	option 2      "Opcao_PerguntarSaida"  -> dialog_medico_saida
	exit_option   "Opcao_Encerrar"
}

dialog_medico_vacina
{
	npc_line      "Medico_ExplicaVacina"
	option 1      "Opcao_ComoFunciona"    -> dialog_medico_vacina_detalhe
	option 2      "Opcao_AceitarVacina"   -> dialog_medico_final
	option 3      "Opcao_Voltar"          -> dialog_medico
	exit_option   "Opcao_Encerrar"
}

dialog_medico_vacina_detalhe
{
	npc_line      "Medico_DetalheVacina"
	option 1      "Opcao_Voltar" -> dialog_medico_vacina
	exit_option   "Opcao_Encerrar"
}

dialog_medico_saida
{
	npc_line      "Medico_ExplicaSaida"
	option 1      "Opcao_Voltar" -> dialog_medico
	exit_option   "Opcao_Encerrar"
}

dialog_medico_final
{
	npc_line      "Medico_Despedida"
	give          "item_antidote"
	option 1      "Opcao_Sair" -> END
}
